# MedAgent v4.0 技术文档

> 最后更新: 2026-04-15

## 项目概述

MedAgent 是一个基于大语言模型的医疗问诊智能体系统，v4.0 版本主要改进：

- **双模式数据合成**: 提供快速模式和质量模式两种合成策略
- **医疗成本标注**: 引入医疗项目/药物的成本标注能力
- **医疗专用Judge**: 使用Baichuan-M3作为医疗&事实特化的评测模型
- **轻量Judge探索**: 测试M2模型作为大批量评测的可行性

## 2. 核心数据结构

### 2.1 MedicalCase (扩展)

v4.0 在原有数据结构基础上新增成本相关字段：

```python
@dataclass
class MedicalItem:
    """医疗数据项 - 用于主观/客观信息"""
    keywords: List[str]   # 触发关键词
    content: str          # 信息内容
    necessity: bool       # 是否必要

@dataclass
class CostInfo:
    """医疗成本信息 (v4.0新增)"""
    items: List[Dict]          # 医疗项目成本
    drugs: List[Dict]          # 药物成本
    total_estimated: float     # 预估总成本
    confidence: str            # 置信度: high/medium/low
    source: str                # 数据来源

@dataclass
class GroundTruth:
    """评测标准答案"""
    diagnosis: List[str]   # 正确诊断
    treatment: List[str]   # 治疗方案
    avoid: List[str]       # 禁忌项

@dataclass
class MedicalCase:
    """标准化医疗病例"""
    case_id: str
    difficulty: str                    # easy/medium/hard
    tags: List[str]                    # 科室、疾病标签
    chief_complaint: str               # 主诉（始终可见）
    subjective: List[MedicalItem]      # 问诊信息
    objective: List[MedicalItem]       # 检查信息
    ground_truth: GroundTruth          # 标准答案
    cost_info: Optional[CostInfo]      # 成本信息 (v4.0新增)
    source: str                        # synthetic/real/seed
```

### 2.2 信息披露机制

| 信息类型 | 获取方式 | 触发机制 |
|---------|---------|---------|
| 主诉 | 始终可见 | 无需触发 |
| 主观信息 (subjective) | ASK工具 | 关键词匹配 |
| 客观信息 (objective) | EXAM工具 | 关键词匹配 |
| 医学知识 | KNOWLEDGE工具 | 语义检索 |

#### 5.2.2 Embedding索引构建

```python
# src/medagent/cost_indexer.py

class CostIndexer:
    """医疗服务+药品价格向量索引器"""

    def __init__(
        self,
        embedding_url: str = "http://localhost:8300/v1",
        db_path: str = "data/cost_db",
    ):
        self.embedding_client = EmbeddingClient(embedding_url)
        self.db = chromadb.PersistentClient(db_path)
        self.collection = self.db.get_or_create_collection(
            name="cost_reference",
            metadata={"hnsw:space": "cosine"}
        )

    def build_index(
        self,
        services_path: str = "data/cost_reference/medical_services.jsonl",
        drugs_path: str = "data/cost_reference/insurance_drugs.jsonl",
    ):
        """
        构建向量索引

        步骤:
        1. 加载医疗服务项目清单和医保药品列表
        2. 合并为统一价格条目列表
        3. 对每个条目生成embedding (使用项目名+描述作为查询文本)
        4. 存入Chroma向量数据库
        """
        # 1. 加载数据
        services = self._load_jsonl(services_path)
        drugs = self._load_jsonl(drugs_path)

        # 2. 合并为统一格式
        all_items = []
        for svc in services:
            all_items.append({
                "id": f"service_{svc['item_code']}",
                "type": "service",
                "name": svc["item_name"],
                "search_text": f"{svc['item_name']} {svc['category']} {svc['description']}",
                "metadata": svc
            })
        for drug in drugs:
            all_items.append({
                "id": f"drug_{drug['drug_code']}",
                "type": "drug",
                "name": drug["drug_name"],
                "search_text": f"{drug['drug_name']} {drug['specification']} {drug['indication']}",
                "metadata": drug
            })

        # 3. 生成embedding并存储
        texts = [item["search_text"] for item in all_items]
        embeddings = self.embedding_client.embed(texts)

        self.collection.add(
            ids=[item["id"] for item in all_items],
            embeddings=embeddings,
            metadatas=[item["metadata"] for item in all_items],
            documents=[item["search_text"] for item in all_items]
        )

        logger.info(f"索引构建完成: {len(all_items)} 条目")
        return len(all_items)

    def _load_jsonl(self, path: str) -> List[Dict]:
        """加载JSONL文件"""
        items = []
        with open(path, 'r', encoding='utf-8') as f:
            for line in f:
                items.append(json.loads(line))
        return items
```

#### 5.3.1 从Agent轨迹抽取项目/药品

```python
# src/medagent/cost_extractor.py

class CostExtractor:
    """从Agent运行轨迹中抽取医疗服务项目和药品"""

    def extract_from_trajectory(self, trajectory: AgentTrajectory) -> Dict:
        """
        从Agent轨迹抽取医疗服务项目和药品

        抽取来源:
        - EXAM工具调用: 检查项目
        - FINAL输出: 治疗方案中的药品和治疗操作
        - ASK工具调用: 可能涉及的检查意向

        Returns:
            {
                "services": ["血常规", "心电图", "CT平扫"],
                "drugs": ["阿莫西林", "布洛芬", "头孢克肟"],
                "procedures": ["静脉注射", "清创缝合"]
            }
        """
        services = []
        drugs = []
        procedures = []

        for step in trajectory.steps:
            action = step.action

            # 从EXAM工具抽取检查项目
            if action.type == "EXAM":
                keywords = action.keywords
                services.extend(self._classify_exam_items(keywords))

            # 从FINAL输出抽取治疗方案
            if action.type == "FINAL":
                treatment = action.treatment
                extracted = self._extract_from_treatment(treatment)
                drugs.extend(extracted["drugs"])
                procedures.extend(extracted["procedures"])

        return {
            "services": list(set(services)),  # 去重
            "drugs": list(set(drugs)),
            "procedures": list(set(procedures))
        }

    def _classify_exam_items(self, keywords: List[str]) -> List[str]:

        top_k: int = 5,
    ):
        self.embedding_client = EmbeddingClient(embedding_url)
        self.db = chromadb.PersistentClient(db_path)
        self.collection = self.db.get_collection("cost_reference")
        self.top_k = top_k

    def retrieve(self, query_items: Dict) -> Dict:
        """
        RAG检索价格条目

        Args:
            query_items: {
                "services": ["血常规", "心电图"],
                "drugs": ["阿莫西林"],
                "procedures": ["静脉注射"]
            }

        Returns:
            {
                "services": [
                    {"query": "血常规", "candidates": [...top_k条目...]},
                    ...
                ],
                "drugs": [...],
                "procedures": [...]
            }
        """
        results = {}

        # 检索服务项目价格
        for service in query_items["services"]:
            query_embedding = self.embedding_client.embed([service])[0]
            candidates = self.collection.query(
                query_embeddings=[query_embedding],
                n_results=self.top_k,
                where={"type": "service"}
            )
            results["services"].append({
                "query": service,
                "candidates": self._format_candidates(candidates)
            })

        # 检索药品价格
        for drug in query_items["drugs"]:
            query_embedding = self.embedding_client.embed([drug])[0]
            candidates = self.collection.query(
                query_embeddings=[query_embedding],
                n_results=self.top_k,
                where={"type": "drug"}
            )
            results["drugs"].append({
                "query": drug,
                "candidates": self._format_candidates(candidates)
            })

        # 检索治疗操作价格 (视为服务)
        for procedure in query_items["procedures"]:
            query_embedding = self.embedding_client.embed([procedure])[0]
            candidates = self.collection.query(
                query_embeddings=[query_embedding],
                n_results=self.top_k,
                where={"type": "service"}
            )
            results["procedures"].append({
                "query": procedure,
                "candidates": self._format_candidates(candidates)
            })

        return results

    def _format_candidates(self, candidates: Dict) -> List[Dict]:
        """格式化检索结果"""
        formatted = []
        for i, (id, metadata, doc, distance) in enumerate(zip(
            candidates["ids"][0],
            candidates["metadatas"][0],
            candidates["documents"][0],
            candidates["distances"][0]
        )):
            formatted.append({
                "id": id,
                "name": metadata.get("item_name") or metadata.get("drug_name"),
                "price_low": metadata.get("price_low"),
                "price_high": metadata.get("price_high"),
                "price_avg": metadata.get("price_avg"),
                "similarity": 1 - distance,  # cosine similarity
                "metadata": metadata
            })
        return formatted
```

#### 5.3.3 Judge模型定价

```python
# src/medagent/cost_judger.py

class CostJudger:
    """使用Judge模型(Baichuan M2/M3)进行最终定价"""

    def __init__(
        self,
        judge_model: str = "Baichuan-M3",
        judge_url: str = "http://localhost:8100/v1",
    ):
        self.llm_client = LLMClient(judge_model, judge_url)

    def judge_prices(self, retrieved_items: Dict) -> CostInfo:
        """
        Judge模型为每个项目/药品给出参考价格

        步骤:
        1. 对于每个query及其Top-K候选
        2. 构建定价prompt，包含候选价格信息
        3. 调用Judge模型，输出最匹配的价格
        4. 汇总所有价格，计算总成本
        """
        item_costs = []
        drug_costs = []

        # 处理服务项目
        for service_result in retrieved_items["services"]:
            cost = self._judge_single_item(
                query=service_result["query"],
                candidates=service_result["candidates"],
                item_type="service"
            )
            item_costs.append(cost)

        # 处理药品
        for drug_result in retrieved_items["drugs"]:
            cost = self._judge_single_item(
                query=drug_result["query"],
                candidates=drug_result["candidates"],
                item_type="drug"
            )
            drug_costs.append(cost)

        # 处理治疗操作
        for procedure_result in retrieved_items["procedures"]:
            cost = self._judge_single_item(
                query=procedure_result["query"],
                candidates=procedure_result["candidates"],
                item_type="service"
            )
            item_costs.append(cost)

        # 汇总
        total_estimated = sum(c["estimated_price"] for c in item_costs + drug_costs)
        confidence = self._calculate_confidence(item_costs + drug_costs)

        return CostInfo(
            items=item_costs,
            drugs=drug_costs,
            total_estimated=total_estimated,
            confidence=confidence,
            source="rag+judge"
        )

    def _judge_single_item(self, query: str, candidates: List[Dict], item_type: str) -> Dict:
        """Judge模型定价单个项目"""

        prompt = self._build_pricing_prompt(query, candidates, item_type)
        response = self.llm_client.generate(prompt)

        # 解析Judge输出
        return self._parse_judge_response(query, candidates, response)

    def _build_pricing_prompt(self, query: str, candidates: List[Dict], item_type: str) -> str:
        """构建定价prompt"""
        type_name = "医疗服务项目" if item_type == "service" else "药品"

        candidates_text = ""
        for i, cand in enumerate(candidates, 1):
            candidates_text += f"""
{i}. {cand['name']}
   - 价格范围: {cand['price_low']}-{cand['price_high']}元
   - 平均价格: {cand['price_avg']}元
   - 相似度: {cand['similarity']:.2f}
"""

        prompt = f"""你是一个医疗经济学专家，需要为以下{type_name}给出参考价格。

**待定价项目/药品**: {query}

**相关参考条目 (Top-{len(candidates)} 相似检索)**:
{candidates_text}

请根据以上参考信息，给出最合适的参考价格。输出格式如下：
```json
{
    "matched_item": "最匹配的条目名称",
    "estimated_price": 数字,
    "confidence": "high/medium/low",
    "reason": "定价理由"
}
```

注意:
- 如果相似度最高的条目与查询完全匹配，使用其price_avg作为参考价格
- 如果存在部分匹配，根据相似度加权调整价格
- 如果所有候选相似度都很低(<0.5)，confidence设为low
"""

        return prompt

    def _parse_judge_response(self, query: str, candidates: List[Dict], response: str) -> Dict:
        """解析Judge模型的定价响应"""
        # 解析JSON输出
        try:
            json_match = re.search(r'\{[^}]+\}', response)
            if json_match:
                parsed = json.loads(json_match.group())
            else:
                # fallback: 使用相似度最高的候选
                best_match = candidates[0] if candidates else None
                parsed = {
                    "matched_item": best_match["name"] if best_match else "未匹配",
                    "estimated_price": best_match["price_avg"] if best_match else 0,
                    "confidence": "low",
                    "reason": "无法解析Judge输出，使用最高相似度候选"
                }
        except Exception as e:
            parsed = {
                "matched_item": candidates[0]["name"] if candidates else "未匹配",
                "estimated_price": candidates[0]["price_avg"] if candidates else 0,
                "confidence": "low",
                "reason": f"解析错误: {str(e)}"
            }

        return {
            "query": query,
            "matched_item": parsed["matched_item"],
            "estimated_price": parsed["estimated_price"],
            "confidence": parsed["confidence"],
            "reason": parsed["reason"],
            "candidates": candidates
        }
```

### 5.4 Judge流程集成

成本标注与主Judge流程整合，作为评测的一部分：

```python
# src/medagent/judger.py (修改)

class Judger:
    """医疗评测器 (集成成本标注)"""

    def __init__(
        self,
        judge_model: str,
        judge_url: str,
        embedding_url: str = "http://localhost:8300/v1",
        cost_db_path: str = "data/cost_db",
    ):
        self.llm_client = LLMClient(judge_model, judge_url)
        self.cost_extractor = CostExtractor()
        self.cost_retriever = CostRetriever(embedding_url, cost_db_path)
        self.cost_judger = CostJudger(judge_model, judge_url)

    def judge(self, trajectory: AgentTrajectory, ground_truth: GroundTruth) -> JudgeResult:
        """
        评测Agent轨迹 (包含成本标注)
        """
        # 1. 原有评测流程 (诊断、治疗、禁忌)
        diagnosis_score = self._judge_diagnosis(trajectory, ground_truth)
        treatment_score = self._judge_treatment(trajectory, ground_truth)
        safety_check = self._check_safety(trajectory, ground_truth)

        # 2. 成本标注 (新增)
        cost_info = self._annotate_cost(trajectory)

        return JudgeResult(
            diagnosis_correct=diagnosis_score.correct,
            diagnosis_score=diagnosis_score.score,
            treatment_score=treatment_score.score,
            total_score=self._calculate_total_score(diagnosis_score, treatment_score, safety_check),
            safety_check=safety_check,
            cost_info=cost_info,  # 新增
            trajectory=trajectory
        )

    def _annotate_cost(self, trajectory: AgentTrajectory) -> CostInfo:
        """成本标注子流程"""
        # 1. 从轨迹抽取项目/药品
        extracted = self.cost_extractor.extract_from_trajectory(trajectory)

        # 2. RAG检索价格候选
        retrieved = self.cost_retriever.retrieve(extracted)

        # 3. Judge模型定价
        cost_info = self.cost_judger.judge_prices(retrieved)

        return cost_info
```

### 6.4 Judge一致性评估

```python
# scripts/judge_consistency_test.py

def evaluate_consistency(results_m3, results_m2):
    """评估两个Judge的一致性"""

    # 1. Pearson相关系数
    scores_m3 = [r.total_score for r in results_m3]
    scores_m2 = [r.total_score for r in results_m2]
    pearson_r = scipy.stats.pearsonr(scores_m3, scores_m2)[0]

    # 2. 分类一致率 (按分数档)
    def get_grade(score):
        if score >= 4.5: return "A"
        elif score >= 3.5: return "B"
        elif score >= 2.5: return "C"
        elif score >= 1.5: return "D"
        else: return "F"

    agreement = sum(
        1 for m3, m2 in zip(results_m3, results_m2)
        if get_grade(m3.total_score) == get_grade(m2.total_score)
    ) / len(results_m3)

    # 3. 诊断正确性一致率
    diag_agreement = sum(
        1 for m3, m2 in zip(results_m3, results_m2)
        if m3.diagnosis_correct == m2.diagnosis_correct
    ) / len(results_m3)

    return {
        "pearson_r": pearson_r,
        "grade_agreement": agreement,
        "diagnosis_agreement": diag_agreement,
        "recommend_m2": pearson_r > 0.85 and agreement > 0.80
    }
```

### 7.1 训练原理

### 7.2 奖励函数设计

```python
# agentic_rl.py

def correctness_reward(completions, ground_truth_diagnosis, ground_truth_treatment):
    """诊断正确性奖励: +1.0诊断 +0.5治疗"""
    reward = 0.0
    for diag in ground_truth_diagnosis:
        if diag.lower() in content:
            reward += 1.0
    for treat in ground_truth_treatment:
        if treat.lower() in content:
            reward += 0.5
    return reward

def avoid_violation_reward(completions, ground_truth_avoid):
    """禁忌违反惩罚: -2.0"""
    for avoid_item in ground_truth_avoid:
        if avoid_item.lower() in content:
            reward -= 2.0
    return reward

def tool_usage_reward(completions):
    """工具使用效率: 2-6次+0.5, 过度/不足惩罚"""
    if 2 <= tool_calls <= 6:
        reward += 0.5
    elif tool_calls > 10:
        reward -= 1.0
    elif tool_calls == 0:
        reward -= 0.5
    return reward

def structure_reward(completions):
    """输出格式奖励: 正确格式+0.6"""
    if "*诊断:" in content:
        reward += 0.3
    if "*治疗:" in content:
        reward += 0.3
    return reward
```
